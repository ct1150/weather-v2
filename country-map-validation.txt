[31m+ function summarize([39m
[31m+   city: CountryWeatherCityViewModel,[39m
[31m+   indices: ReadonlyArray<number>,[39m
[31m+   filters: WeatherFilters,[39m
[31m+   locale: ExplorerLocale,[39m
[31m+ ): CitySummary {[39m

[31m+   const rainAmounts = numericValues(days.map((day) => day.weather.precipitationMm));[39m
[31m+   const minimums = numericValues(days.map((day) => day.weather.temperatureMin));[39m
[31m+   const maximums = numericValues(days.map((day) => day.weather.temperatureMax));[39m
[31m+   const winds = numericValues(days.map((day) => day.weather.windSpeedMax));[39m
[31m+   const dryDays = days.filter((day) => {[39m
[31m+     const chance = day.weather.rainProbability;[39m
[31m+     const amount = day.weather.precipitationMm;[39m
[31m+     return amount !== undefined && amount !== null[39m
[31m+       ? amount <= 2.5 || ((chance ?? 100) <= 45 && amount < 5)[39m
[31m+       : chance !== null && chance <= 45;[39m
[31m+   }).length;[39m
[31m+   const maxRain = rainValues.length > 0 ? Math.max(...rainValues) : null;[39m
[31m+   const totalRainMm =[39m
[31m+     rainAmounts.length > 0[39m
[31m+       ? Math.round(rainAmounts.reduce((total, value) => total + value, 0) * 10) / 10[39m
[31m+       : null;[39m
[31m+   const temperatureMin = minimums.length > 0 ? Math.min(...minimums) : null;[39m
[31m+   const temperatureMax = maximums.length > 0 ? Math.max(...maximums) : null;[39m
[31m+   const maxWind = winds.length > 0 ? Math.round(Math.max(...winds)) : null;[39m
[31m+   const risk: Risk =[39m
[31m+     days.length === 0 || (totalRainMm === null && maxRain === null)[39m
[31m+       ? "unknown"[39m
[31m+       : dryDays === days.length && (totalRainMm ?? 0) <= days.length * 2.5[39m
[31m+         ? "good"[39m
[31m+         : dryDays >= Math.ceil(days.length / 2) || (totalRainMm ?? Infinity) <= days.length * 8[39m
[31m+           ? "mixed"[39m
[31m+           : "wet";[39m
[31m+   const metrics = { maxRain, maxWind, temperatureMin, temperatureMax };[39m
[31m+   const reasons = filterReasons(metrics, filters, locale);[39m
[31m+   return {[39m
[31m+     city,[39m
[31m+     days,[39m
[31m+     dryDays,[39m
[31m+     maxRain,[39m
[31m+     totalRainMm,[39m
[31m+     temperatureMin,[39m
[31m+     temperatureMax,[39m
[31m+     maxWind,[39m
[31m+     risk,[39m
[31m+     symbol: summarySymbol(days, risk),[39m
[31m+     filtered: reasons.length > 0,[39m
[31m+     filterReasons: reasons,[39m
[31m+   };[39m
[31m+ }[39m
